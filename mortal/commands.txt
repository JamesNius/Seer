# 2025/2026 纯 JSON 转 gzip
python -c "
import gzip, os
for year in ['2025','2026']:
    d = f'E:/天凤训练数据/{year}'
    for f in os.listdir(d):
        if not f.endswith('.mjson'): continue
        path = os.path.join(d, f)
        with open(path,'rb') as fh:
            if fh.read(2) == b'\x1f\x8b': continue
        tmp = path+'.gz_tmp'
        with open(path,'rb') as src, gzip.open(tmp,'wb') as dst:
            dst.write(src.read())
        os.replace(tmp, path)
    print(year, 'done')
